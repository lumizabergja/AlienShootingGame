cmake_minimum_required(VERSION 3.25)
project(FrameWarpCustom LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
if(NOT DEFINED DLSS_SDK)
  message(FATAL_ERROR "DLSS_SDK must point to the NVIDIA/DLSS source root")
endif()
add_executable(FrameWarpCustom WIN32 src/main.cpp)
set(CPPWINRT_INC "C:/Program Files (x86)/Windows Kits/10/Include/10.0.26100.0/cppwinrt")
target_include_directories(FrameWarpCustom PRIVATE "${DLSS_SDK}/include" "${CPPWINRT_INC}")
target_link_directories(FrameWarpCustom PRIVATE "${DLSS_SDK}/lib/Windows_x86_64/x64")
target_link_libraries(FrameWarpCustom PRIVATE nvsdk_ngx_s d3d11 d3d12 d3dcompiler dxgi dwmapi comctl32 runtimeobject WindowsApp)
target_compile_options(FrameWarpCustom PRIVATE /EHsc /W3 /utf-8 /await:strict)
set_property(TARGET FrameWarpCustom PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
