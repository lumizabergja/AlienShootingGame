cmake_minimum_required(VERSION 3.20)
project(FrameGeneration LANGUAGES CXX)
add_executable(FrameGeneration WIN32 src/main.cpp src/engine.cpp src/runtime.cpp src/resolve.cpp src/nvfbc_capture.cpp src/preset_c.cpp src/diagnostics.cpp src/nvofa_motion.cpp)
target_compile_features(FrameGeneration PRIVATE cxx_std_17)
target_compile_definitions(FrameGeneration PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_include_directories(FrameGeneration PRIVATE vendor/streamline vendor/nvapi vendor/nvof)
target_link_libraries(FrameGeneration PRIVATE d3d9 d3d12 dxgi d3dcompiler dwmapi user32 gdi32 comctl32 wintrust crypt32 ole32 uuid avrt version)
if(MSVC)
 target_compile_options(FrameGeneration PRIVATE /O2 /Ob3 /Gw /Gy /W4 /EHsc)
 set_property(TARGET FrameGeneration PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
 set_property(TARGET FrameGeneration PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
 target_compile_options(FrameGeneration PRIVATE -O2 -Wall -Wextra)
 target_link_options(FrameGeneration PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
