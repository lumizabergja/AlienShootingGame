cmake_minimum_required(VERSION 3.24)
project(FreshMotionFG LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "FreshMotionFG builds only on Windows")
endif()
if(NOT DEFINED STREAMLINE_INCLUDE)
  message(FATAL_ERROR "Set STREAMLINE_INCLUDE to NVIDIA Streamline's include directory")
endif()
add_executable(FreshMotionFG WIN32 src/main.cpp src/nvfbc_capture.cpp)
target_include_directories(FreshMotionFG PRIVATE "${STREAMLINE_INCLUDE}")
target_compile_definitions(FreshMotionFG PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(FreshMotionFG PRIVATE d3d9 d3d12 dxgi d3dcompiler dcomp user32 gdi32 shcore comctl32)
target_compile_options(FreshMotionFG PRIVATE /W3 /EHsc)
