cmake_minimum_required(VERSION 3.24)
project(NvFastConfidence LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "Windows x64 is required")
endif()
add_executable(NvFastConfidence WIN32 src/main.cpp src/engine.cpp src/capture.cpp)
target_include_directories(NvFastConfidence PRIVATE "${STREAMLINE_INCLUDE}" vendor/nvof src)
target_compile_definitions(NvFastConfidence PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(NvFastConfidence PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(NvFastConfidence PRIVATE d3d12 dxgi d3d9 d3dcompiler user32 gdi32 shell32 wintrust crypt32)
add_executable(ShaderCheck tests/shader_check.cpp)
target_compile_definitions(ShaderCheck PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(ShaderCheck PRIVATE src)
target_link_libraries(ShaderCheck PRIVATE d3dcompiler d3d12 dxgi)
enable_testing()
add_test(NAME ShaderCompileAndGpuConversion COMMAND ShaderCheck)
