cmake_minimum_required(VERSION 3.20)
project(NvShaderFresh LANGUAGES CXX)
add_executable(NvShaderFresh WIN32 src/main.cpp src/nvfbc_capture.cpp)
target_compile_features(NvShaderFresh PRIVATE cxx_std_17)
target_compile_definitions(NvShaderFresh PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_include_directories(NvShaderFresh PRIVATE vendor/streamline)
target_link_libraries(NvShaderFresh PRIVATE d3d9 d3d12 dxgi d3dcompiler user32 gdi32 comctl32 ole32 uuid avrt)
if(MSVC)
  target_compile_options(NvShaderFresh PRIVATE /O2 /W4 /EHsc)
  set_property(TARGET NvShaderFresh PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
