cmake_minimum_required(VERSION 3.20)
project(NvShader LANGUAGES CXX)
add_executable(NvShader WIN32 src/main.cpp src/nvfbc_capture.cpp)
target_compile_features(NvShader PRIVATE cxx_std_17)
target_compile_definitions(NvShader PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_include_directories(NvShader PRIVATE vendor/streamline)
target_link_libraries(NvShader PRIVATE d3d9 d3d12 dxgi d3dcompiler user32 gdi32 comctl32 ole32 uuid avrt)
if(MSVC)
  target_compile_options(NvShader PRIVATE /O2 /W4 /EHsc)
  set_property(TARGET NvShader PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
